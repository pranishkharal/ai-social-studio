Pillow>=10.0
numpy>=1.26
# OpenCV 5.x removed CascadeClassifier entirely — or so it was believed.
# Verified 2026-11-21 in this VM: mediapipe 1.2.3 pulls
# opencv-contrib-python 5.1.0.95, and CascadeClassifier still loads the
# repo's haarcascade file and runs detectMultiScale fine on 5.1.0.
# The <5 pin is kept anyway for installs WITHOUT mediapipe: it is the
# combination the test suite and the Haar path were validated against,
# and opencv 5.x behaviour outside CascadeClassifier is untested here.
# NOTE: mediapipe (below) depends on opencv-contrib-python (unpinned,
# currently 5.x), which overwrites the cv2 files from this package —
# with mediapipe installed, cv2 reports 5.x. That is expected and the
# Haar path was verified working on 5.1.0.95.
opencv-python-headless>=4.9,<5

# Face tracking (speaker_tracker.py): the primary MediaPipe backend.
# Tasks API (mediapipe 1.x has no mp.solutions); the BlazeFace model
# (~230 KB) auto-downloads into models/ on first use. On Linux the
# Tasks API also needs the system EGL/GLES libraries (libegl1,
# libgles2) — without them MediaPipe init fails and the tracker falls
# back to Haar (see the README's face-tracking section).
mediapipe>=1.0

# AI: transcription + story generation (ai_story_engine.py)
faster-whisper>=1.0
# faster-whisper 1.2.1 (the latest release) passes
# metadata_errors="ignore" to av.open(); PyAV 19 removed that keyword,
# so transcription fails outright on a fresh install with av>=19.
# av 15-18 still accept it and ship wheels for Windows, macOS and Linux
# (av 14.x has no wheel for several current platforms — do not pin lower).
av>=15,<19
google-genai>=1.0

# Browser automation for TikTok upload (screen_controller.py)
playwright>=1.44

# Live chat hype via Kick's Pusher websocket (chat_hype.py)
websocket-client>=1.8

# YouTube Shorts uploader (youtube_uploader.py)
google-api-python-client>=2.0
google-auth-oauthlib>=1.2
